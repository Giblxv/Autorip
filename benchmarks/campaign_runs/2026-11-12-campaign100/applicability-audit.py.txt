"""Untimed branch-coverage check for candidate shortcuts."""
import json
from pathlib import Path
import sys
import threading

root=Path('/workspace/Autorip-campaign100')
review=Path('/workspace/autorip-review/campaign100-2026-11-12')
sys.path.insert(0,str(root/'benchmarks'))
import rip_campaign as c
from rip_wide_candidates import CANDIDATES as wide
from rip_rgba_candidates import CANDIDATES as rgba
from rip_io_candidates import CANDIDATES as io

base={k:(root/p).read_text() for k,p in c.FILES.items()}
assert c.digest(base)=='4931ab0acb27b5753a0a78e2aaef4dbd1ed5306096afc4b61ef9de1bec8dedd5'
folder=review/'applicability';folder.mkdir(exist_ok=True)
cases=[x for x in c.make_cases(folder,'/workspace/Autorip/Sublinks_DTF_E602 LM720x2400 8Pass .icc','large-io-v3') if x['timed']]
proposals={}
for item in wide+rgba+io:proposals.setdefault(item[0],item)
results={}
for name in ('subpixel-morph','halftone-full','rgba-copied-in-place'):
    candidate=proposals[name][2](base)
    modules=c.load(candidate,folder/name,name)
    markers={}
    for key,source in candidate.items():
        for index,line in enumerate(source.splitlines(),1):
            kind=('subpixel-predicate' if 'radius < min(sampling)' in line else
                  'full-channel-return' if 'return np.full(v.shape, 3' in line else
                  'rgba-copy-branch' if 'copied = source.copy()' in line else None)
            if kind:markers[(str(folder/name/f'{key}.py'),index)]=kind
    entries=[]
    for case in cases:
        seen=set();active=set()
        def trace(frame,event,arg):
            if event=='line':
                kind=markers.get((frame.f_code.co_filename,frame.f_lineno))
                if kind:
                    seen.add(kind)
                    if kind!='subpixel-predicate' or (frame.f_locals['radius'] and frame.f_locals['radius']<min(frame.f_locals['sampling'])):
                        active.add(kind)
            return trace
        sys.settrace(trace);threading.settrace(trace)
        try:c.render(c.new_rip(modules['engine'],case),case,folder/'coverage.PRT',True)
        finally:sys.settrace(None);threading.settrace(None)
        entries.append(dict(case=case['name'],seen=sorted(seen),active=sorted(active)))
    results[name]=dict(active=any(x['active'] for x in entries),cases=entries)
(review/'applicability.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps({name:value['active'] for name,value in results.items()},indent=2))
