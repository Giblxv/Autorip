import hashlib,json,subprocess,sys
from pathlib import Path
if not __debug__:
    raise RuntimeError('Run campaign safety checks without Python -O')

repo='Giblxv/Autorip';tag='v0.8.25';expected=sys.argv[1];run_ids=[int(x) for x in sys.argv[2:]]
def api(path):
 return json.loads(subprocess.check_output(['gh','api',f'repos/{repo}/{path}']))
r=api(f'releases/tags/{tag}');latest=api('releases/latest')
assert not r['draft'] and not r['prerelease'] and r['published_at']
assert latest['id']==r['id'] and r['tag_name']==tag
remote=subprocess.check_output(['git','ls-remote','origin',f'refs/tags/{tag}^{{}}'],text=True).split()[0]
assert remote==expected
assets={a['name']:a for a in r['assets']}
assert set(assets)=={'AutoRIP.exe','PrintExpAgent.exe','AutoRIP_Setup.exe','SHA256SUMS.txt'}
folder=Path('/workspace/autorip-review/campaign100-2026-11-12/release-v0.8.25')
folder.mkdir(exist_ok=True)
subprocess.run(['gh','release','download',tag,'--repo',repo,'--dir',str(folder),'--clobber'],check=True)
manifest={}
for line in (folder/'SHA256SUMS.txt').read_text().splitlines():
 if not line.strip():continue
 sha,name=line.split(maxsplit=1)
 name=name.replace('\\','/').rsplit('/',1)[-1]
 assert name not in manifest, 'Duplicate manifest asset'
 assert len(sha)==64 and all(c in '0123456789abcdef' for c in sha), 'Invalid SHA256'
 manifest[name]=sha
assert set(manifest)=={'AutoRIP.exe','PrintExpAgent.exe','AutoRIP_Setup.exe'}
verified=[]
for name,a in assets.items():
 path=folder/name
 with path.open('rb') as stream:digest=hashlib.file_digest(stream,'sha256').hexdigest()
 assert path.stat().st_size==a['size'],name
 assert a['digest']=='sha256:'+digest,name
 if name in manifest:assert manifest[name]==digest,name
 verified.append(dict(name=name,bytes=path.stat().st_size,sha256=digest))
runs=[]
for rid in run_ids:
 run=api(f'actions/runs/{rid}');assert run['conclusion']=='success'
 jobs=api(f'actions/runs/{rid}/jobs')['jobs']
 if rid==run_ids[-1]:
  assert run['head_sha']==expected
  build=next(j for j in jobs if j['name']=='build')
  for name in ['Conferir versao da tag e do instalador','Run regression tests before Windows packaging','Smoke test do executavel empacotado','Smoke test installed executable']:
   assert next(s for s in build['steps'] if s['name']==name)['conclusion']=='success',name
 runs.append(dict(id=rid,sha=run['head_sha'],conclusion=run['conclusion'],url=run['html_url'],jobs=[dict(name=j['name'],conclusion=j['conclusion'],steps=[dict(name=s['name'],conclusion=s['conclusion']) for s in j['steps']]) for j in jobs]))
result=dict(tag=tag,commit=expected,url=r['html_url'],latest_confirmed=True,manifest_and_api_sha256_verified=True,assets=verified,ci=runs)
(folder.parent/'release-verification.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k!='ci'},indent=2))
