"""Fast offline checks of the autonomous publication barriers."""
import importlib.util
import json
from pathlib import Path
import tempfile
import unittest
from types import SimpleNamespace
from unittest.mock import patch


class PublicationGates(unittest.TestCase):
    def setUp(self):
        path=Path(__file__).with_name('finalize.py')
        spec=importlib.util.spec_from_file_location('campaign_finalizer_under_test',path)
        self.module=importlib.util.module_from_spec(spec);spec.loader.exec_module(self.module)
        self.folder=tempfile.TemporaryDirectory()
        self.addCleanup(self.folder.cleanup)
        self.module.REVIEW=Path(self.folder.name)

    def checks(self,failed=None):
        names=['build','Testes ubuntu-latest / Python 3.10','Testes ubuntu-latest / Python 3.12',
               'Testes windows-2022 / Python 3.10','Testes windows-2022 / Python 3.12']
        return [dict(name=name,status='COMPLETED',conclusion='FAILURE' if name==failed else 'SUCCESS',
                     detailsUrl=f'https://github.com/Giblxv/Autorip/actions/runs/{10 if name=="build" else 20}/job/1')
                for name in names]

    def test_exact_head_and_all_required_checks_allow_progress(self):
        self.module.command=lambda *args:json.dumps(dict(state='OPEN',headRefOid='approved',statusCheckRollup=self.checks()))
        self.module.api=lambda path:dict(head_sha='approved')
        self.assertEqual(self.module.wait_checks('approved'),[20,10])

    def test_changed_pr_head_blocks(self):
        self.module.command=lambda *args:json.dumps(dict(state='OPEN',headRefOid='someone-else',statusCheckRollup=self.checks()))
        with self.assertRaisesRegex(AssertionError,'PR changed'):
            self.module.wait_checks('approved')

    def test_failed_installer_build_blocks(self):
        self.module.command=lambda *args:json.dumps(dict(state='OPEN',headRefOid='approved',statusCheckRollup=self.checks('build')))
        with self.assertRaisesRegex(AssertionError,'CI failed'):
            self.module.wait_checks('approved')

    def test_failed_tag_does_not_claim_release(self):
        self.module.command=lambda *args:json.dumps([dict(databaseId=30,headSha='approved',
            workflowName='Build Windows',status='completed',conclusion='failure')])
        with self.assertRaisesRegex(AssertionError,'Tag build failed'):
            self.module.wait_tag('approved')

    def test_existing_version_is_never_overwritten(self):
        def command(*args):
            if args[:3]==('git','branch','--show-current'):return self.module.BRANCH
            if args[:3]==('gh','repo','view'):return self.module.REPO
            if args[:3]==('git','ls-remote','origin'):return 'existing-tag'
            raise AssertionError('Unexpected command: '+repr(args))
        self.module.command=command
        self.module.api=lambda path:dict(object=dict(sha=self.module.BASE))
        (self.module.REVIEW/'campaign.json').write_text(json.dumps(dict(required_without_gain=100,deadline='2026-11-13T22:00:00-03:00')))
        with self.assertRaisesRegex(AssertionError,'never overwrite'):
            self.module.readiness()

    def test_unrelated_concurrent_main_is_preserved(self):
        def command(*args):
            if args[:3]==('git','branch','--show-current'):return self.module.BRANCH
            if args[:3]==('gh','repo','view'):return self.module.REPO
            if args[:3]==('git','fetch','origin'):return ''
            raise AssertionError('Unexpected command: '+repr(args))
        self.module.command=command
        self.module.api=lambda path:dict(object=dict(sha='changed')) if path=='git/ref/heads/main' else []
        (self.module.REVIEW/'campaign.json').write_text(json.dumps(dict(required_without_gain=100,deadline='2026-11-13T22:00:00-03:00')))
        (self.module.REVIEW/'framework-head.txt').write_text('framework')
        with self.assertRaisesRegex(AssertionError,'unrelated main'):
            self.module.readiness()

    def known_merge(self,matching_tree):
        def command(*args):
            if args[:3]==('git','branch','--show-current'):return self.module.BRANCH
            if args[:3]==('gh','repo','view'):return self.module.REPO
            if args[:3]==('git','fetch','origin'):return ''
            if args[:3]==('git','ls-remote','origin'):return ''
            if args[:2]==('git','rev-parse'):
                return 'known-tree' if matching_tree or args[2]=='known^{tree}' else 'other-tree'
            raise AssertionError('Unexpected command: '+repr(args))
        self.module.command=command
        self.module.api=lambda path:(dict(object=dict(sha='changed')) if path=='git/ref/heads/main' else
                                    [dict(merged_at='now',head=dict(ref=self.module.BRANCH,sha='known'))])
        (self.module.REVIEW/'campaign.json').write_text(json.dumps(dict(required_without_gain=100,deadline='2026-11-13T22:00:00-03:00')))
        (self.module.REVIEW/'framework-head.txt').write_text('framework')
        for name in ('audit.py','runner.py','verify_release26.py'):(self.module.REVIEW/name).write_text('pass\n')
        with patch.object(self.module.subprocess,'run',return_value=SimpleNamespace(returncode=0)):
            return self.module.readiness()

    def test_known_reviewed_merge_can_advance_main(self):
        self.assertEqual(self.known_merge(True)['main'],'changed')

    def test_own_pr_with_unknown_merged_tree_still_blocks(self):
        with self.assertRaisesRegex(AssertionError,'unrelated main'):
            self.known_merge(False)


if __name__=='__main__':unittest.main()
