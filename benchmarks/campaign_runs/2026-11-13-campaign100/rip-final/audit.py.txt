"""Independent source/count/output audit; never converts a deadline into 100 rounds."""
from datetime import datetime
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import tempfile
import zipfile

if not __debug__:
    raise RuntimeError('Run campaign safety checks without Python -O')

ROOT=Path('/workspace/Autorip-campaign101')
REVIEW=Path('/workspace/autorip-review/campaign100-2026-11-13')


def audit():
    sys.path.insert(0,str(ROOT/'benchmarks'))
    import rip_campaign as c
    c.DEADLINE=None
    from rip_tuning_candidates import execution_plan
    work=REVIEW/'rip'
    report=json.loads((work/'report.json').read_text())
    if report.get('cpu_affinity') and hasattr(os,'sched_setaffinity'):
        os.sched_setaffinity(0,report['cpu_affinity'])
    meta=json.loads((REVIEW/'campaign.json').read_text())
    supervisor=json.loads((REVIEW/'status.json').read_text())
    assert report['required_without_gain']==meta['required_without_gain']==100
    count=0
    assert meta['initial_rip_without_gain']==report['initial_without_gain']==0
    raw_pairs=set(); semantic_pairs=set(); plan_pairs=set()
    initial={k:(work/'initial'/f'{k}.py').read_text() for k in c.FILES}
    leader=initial
    accepted_semantics={c.semantic_digest(initial)}
    for index,row in enumerate(report['rounds'],1):
        assert row['round']==index and row['reason'] not in ('exception','infrastructure-error')
        assert row['leader_before']==c.digest(leader)
        folder=work/f"round-{index:03}-{row['candidate']}"
        candidate={k:(folder/f'{k}.py').read_text() for k in c.FILES}
        assert c.digest(candidate)==row['candidate_hash']
        raw=(c.digest(leader),c.digest(candidate))
        sem=(c.semantic_digest(leader),c.semantic_digest(candidate))
        assert raw not in raw_pairs and sem not in semantic_pairs and sem[0]!=sem[1]
        assert sem[1] not in accepted_semantics
        assert sem==(row['leader_semantic_hash'],row['candidate_semantic_hash'])
        raw_pairs.add(raw);semantic_pairs.add(sem)
        if 'execution_plan_sha256' in row:
            key=(row['leader_before'],row['execution_plan_sha256'])
            assert key not in plan_pairs;plan_pairs.add(key)
        if row['accepted']:
            check=row['regression_validation']
            assert check['passed'] and check['returncode']==0
            assert set(row['heldout_reference']['scenarios'])=={'rgba_gamma_flat','icc_single_cpu'}
            assert all(s['speedup']>=1/1.06 for s in row['heldout_reference']['scenarios'].values())
            assert hashlib.sha256(Path(check['log']).read_bytes()).hexdigest()==check['sha256']
            leader=candidate;count=0
            accepted_semantics.add(sem[1])
        else:count+=1
        assert count==row['consecutive_without_gain']
    assert report['consecutive_without_gain']==count and report['leader_hash']==c.digest(leader)
    checkout={k:(ROOT/p).read_text() for k,p in c.FILES.items()}
    assert c.digest(checkout)==c.digest(leader)
    assert all(hashlib.sha256((ROOT/p).read_bytes()).hexdigest()==sha for p,sha in report['frozen_inputs'].items())
    if count>=100:
        stop='consecutive-without-gain'
        assert not any(row['accepted'] for row in report['rounds'][-100:])
    else:
        stop='deadline'
        assert report.get('stop_reason')=='deadline' or supervisor.get('external_interrupt')=='deadline-watchdog'
        stopped=report.get('stopped_at_utc') or supervisor['search_stopped_at_utc']
        assert datetime.fromisoformat(stopped).timestamp()>=datetime.fromisoformat(meta['deadline']).timestamp()
    history=set();semhistory=set()
    for entry in report['history_sources']:
        path=Path(entry['path']); data=path.read_bytes()
        assert hashlib.sha256(data).hexdigest()==entry['sha256']
        archived=path.parent/'source-snapshots.zip'
        if entry.get('archive_sha256'):
            assert hashlib.sha256(archived.read_bytes()).hexdigest()==entry['archive_sha256']
        past=json.loads(data)
        history.update((x['leader_before'],x['candidate_hash']) for x in past.get('rounds',[])
                       if 'leader_before' in x and 'candidate_hash' in x
                       and x.get('reason') not in ('exception','infrastructure-error'))
        semhistory.update(c.historical_semantic_pairs(past,path))
    assert not (raw_pairs & history) and not (semantic_pairs & semhistory)
    historical_plans={(x['leader_hash'],x['execution_plan_sha256']) for x in report['historical_execution_plans']}
    assert not (plan_pairs & historical_plans)
    icc=Path('/workspace/Autorip/Sublinks_DTF_E602 LM720x2400 8Pass .icc')
    icc_sha=hashlib.sha256(icc.read_bytes()).hexdigest()
    assert icc_sha=='a86396b8ca0b1799096359fec6ac79e39db454fed71ba6dd2eed24d53506d6a5'
    with tempfile.TemporaryDirectory(prefix='campaign100-final-') as directory:
        folder=Path(directory)
        original=c.load(initial,folder/'baseline','audit_base')
        final=c.load(leader,folder/'final','audit_final')
        assert c.micro_correct(original,final) is None
        cases=c.make_cases(folder,str(icc),report['suite']);timed=[x for x in cases if x['timed']]
        hashes={}
        for case in cases:
            a=c.render(c.new_rip(original['engine'],case),case,folder/'out.PRT',True)
            b=c.render(c.new_rip(final['engine'],case),case,folder/'out.PRT',True)
            assert a==b,case['name']
            hashes[case['name']]=dict(baseline=a,final=b,equal=True)
        for row in report['rounds']:
            if 'execution_plan_sha256' not in row:continue
            candidate={k:(work/f"round-{row['round']:03}-{row['candidate']}"/f'{k}.py').read_text() for k in c.FILES}
            module=c.load(candidate,folder/'plan',f"audit_plan_{row['round']}")
            checksum,plan=execution_plan(candidate,module['engine'],timed)
            assert checksum==row['execution_plan_sha256'] and plan==row['execution_plan']
        recomputed_history=[]
        for entry in report['history_sources']:
            path=Path(entry['path']);past=json.loads(path.read_text())
            for row,snapshot in c.historical_scheduling_candidates(past,path,initial):
                module=c.load(snapshot,folder/'historical-plan','audit_historical_plan')
                actual,_=execution_plan(snapshot,module['engine'],timed)
                recomputed_history.append(dict(path=str(path),round=row['round'],candidate_hash=row['candidate_hash'],
                    leader_hash=c.digest(initial),execution_plan_sha256=actual))
        assert recomputed_history==report['historical_execution_plans']
    assert len(hashes)==30
    result=dict(verified=True,stop_reason=stop,driver_stop_reason=report.get('stop_reason'),
                rounds=len(report['rounds']),phase_rounds=len(report['rounds']),
                prefix_rounds=len(meta['prefix_rounds']),prefix_rip_rounds=0,
                consecutive_without_gain=count,required_without_gain=100,convergence_reached=count>=100,
                rip_accepted=sum(x['accepted'] for x in report['rounds']),partial_rounds_counted=False,
                source_hash=c.digest(leader),icc_sha256=icc_sha,prt_sha256=hashes,
                historical_duplicates=0,semantic_duplicates=0,execution_plan_duplicates=0)
    (REVIEW/'audit-result.json').write_text(json.dumps(result,indent=2)+'\n')
    out=ROOT/'benchmarks/campaign_runs/2026-11-13-campaign100/rip-final'
    out.mkdir(parents=True,exist_ok=True)
    shutil.copy2(work/'report.json',out/'report.json')
    shutil.copy2(REVIEW/'audit-result.json',out/'verification.json')
    shutil.copy2(REVIEW/'rip.log',out/'progress.log')
    manifest={}
    with zipfile.ZipFile(out/'source-snapshots.zip','w',zipfile.ZIP_DEFLATED) as archive:
        for directory in [work/'initial']+sorted(work.glob('round-*')):
            for key in c.FILES:
                path=directory/f'{key}.py'
                data=path.read_bytes();name=str(path.relative_to(work))
                archive.writestr(name,data);manifest[name]=hashlib.sha256(data).hexdigest()
        for key,source in leader.items():
            name=f'final_modules/{key}.py';data=source.encode()
            archive.writestr(name,data);manifest[name]=hashlib.sha256(data).hexdigest()
    (out/'source-manifest.json').write_text(json.dumps(manifest,indent=2)+'\n')
    with zipfile.ZipFile(out/'source-snapshots.zip') as archive:
        assert set(archive.namelist())==set(manifest)
        assert all(hashlib.sha256(archive.read(name)).hexdigest()==sha for name,sha in manifest.items())
    for filename in ('rip_campaign.py','rip_candidates.py','rip_wide_candidates.py',
                     'rip_rgba_candidates.py','rip_io_candidates.py','rip_tuning_candidates.py','rip_color_candidates.py'):
        data=subprocess.check_output(['git','show',f"{report['base_commit']}:benchmarks/{filename}"],cwd=ROOT)
        (out/(filename+'.txt')).write_bytes(data)
    assert hashlib.sha256((out/'rip_campaign.py.txt').read_bytes()).hexdigest()==report['benchmark_sha256']
    shutil.copy2(__file__,out/'audit.py.txt')
    (out/'README.md').write_text(f"# RIP final\n\nParada: `{stop}`; rodadas concluídas: {result['rounds']}; aceites RIP: {result['rip_accepted']}; sequência sem ganho: {count}/100. Convergência: {result['convergence_reached']}. Rodadas parciais, configurações equivalentes e erros não contam. O relatório bruto é preservado, inclusive sua flag original de conclusão. Os módulos finais são derivados do líder auditado. Foram conferidos hashes/ASTs históricos, planos de execução, regressões de cada aceite, checkout, ZIP e 30 PRTs idênticos. Pico do benchmark é tracemalloc, sem todo o RSS nativo.\n")
    print(json.dumps({k:v for k,v in result.items() if k!='prt_sha256'},indent=2))
    return result


if __name__=='__main__':audit()
