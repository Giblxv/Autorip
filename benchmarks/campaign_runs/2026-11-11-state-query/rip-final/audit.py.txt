"""Audit a completed or deadline-stopped campaign without claiming convergence."""
import hashlib
import json
from pathlib import Path
import shutil
import subprocess
import sys
import tempfile
import zipfile

root = Path('/workspace/Autorip')
review = Path('/workspace/autorip-review/state-query-2026-11-11')
work = review / 'rip-v3'
sys.path.insert(0, str(root / 'benchmarks'))
from rip_campaign import FILES, digest, load, make_cases, micro_correct, new_rip, render

report = json.loads((work / 'report.json').read_text())
assert report.get('stop_reason') in ('deadline', 'consecutive-without-gain')
initial = {k: (work / 'initial' / (k + '.py')).read_text() for k in FILES}
leader = initial
seen = set()
count = 0
for index, record in enumerate(report['rounds'], 1):
    assert record['round'] == index
    assert record['reason'] not in ('exception', 'infrastructure-error')
    assert record['leader_before'] == digest(leader)
    folder = work / f"round-{index:03}-{record['candidate']}"
    candidate = {k: (folder / (k + '.py')).read_text() for k in FILES}
    assert digest(candidate) == record['candidate_hash']
    key = (record['leader_before'], record['candidate_hash'])
    assert key not in seen
    seen.add(key)
    if record['accepted']:
        leader = candidate
        count = 0
    else:
        count += 1
    assert record['consecutive_without_gain'] == count
assert report['consecutive_without_gain'] == count
assert report['leader_hash'] == digest(leader)
checkout = {k: (root / path).read_text() for k, path in FILES.items()}
assert digest(checkout) == digest(leader)
if report['stop_reason'] == 'consecutive-without-gain':
    assert report['complete'] and count >= 25
else:
    assert report.get('interrupted_round_counted') is False
history = set()
for source in report.get('history_sources', []):
    data = Path(source['path']).read_bytes()
    assert hashlib.sha256(data).hexdigest() == source['sha256']
    past = json.loads(data)
    history.update((r['leader_before'], r['candidate_hash']) for r in past.get('rounds', [])
                   if 'leader_before' in r and 'candidate_hash' in r
                   and r.get('reason') not in ('infrastructure-error', 'exception'))
assert not (history & seen)
icc = root / 'Sublinks_DTF_E602 LM720x2400 8Pass .icc'
with tempfile.TemporaryDirectory(prefix='autorip-deadline-audit-') as directory:
    folder = Path(directory)
    old = load(initial, folder / 'old', 'audit-old')
    new = load(leader, folder / 'new', 'audit-new')
    assert micro_correct(old, new) is None
    hashes = {}
    for case in make_cases(folder, str(icc), report['suite']):
        a = render(new_rip(old['engine'], case), case, folder / 'output.PRT', True)
        b = render(new_rip(new['engine'], case), case, folder / 'output.PRT', True)
        assert a == b, case['name']
        hashes[case['name']] = {'baseline': a, 'final': b, 'equal': True}
assert len(hashes) == 30
result = dict(verified=True, rounds=len(report['rounds']), consecutive_without_gain=count,
              required_without_gain=25, convergence_reached=count >= 25,
              stop_reason=report['stop_reason'], accepted=sum(r['accepted'] for r in report['rounds']),
              checkout_checked=True, final_source_hash=digest(leader),
              partial_round_counted=False, historical_duplicates=0,
              icc_sha256=hashlib.sha256(icc.read_bytes()).hexdigest(), prt_sha256=hashes)
(work / 'deadline-verification.json').write_text(json.dumps(result, indent=2) + '\n')
out = root / 'benchmarks/campaign_runs/2026-11-11-state-query/rip-final'
out.mkdir(parents=True, exist_ok=True)
shutil.copy2(work / 'report.json', out / 'report.json')
shutil.copy2(work / 'deadline-verification.json', out / 'verification.json')
for name in ['rip-v3-progress.log', 'v3-restart.json', 'v3-driver-tests.log']:
    shutil.copy2(review / name, out / name)
manifest = {}
folders = ['initial'] + [f"round-{r['round']:03}-{r['candidate']}" for r in report['rounds']]
partial_folders = sorted(p.name for p in work.glob('round-*') if p.is_dir() and p.name not in folders)
folders += partial_folders
with zipfile.ZipFile(out / 'source-snapshots.zip', 'w', compression=zipfile.ZIP_DEFLATED) as archive:
    for folder in folders:
        for k in FILES:
            name = f'{folder}/{k}.py'
            data = (work / name).read_bytes()
            manifest[name] = hashlib.sha256(data).hexdigest()
            archive.writestr(name, data)
    for k, source in leader.items():
        name = f'final_modules/{k}.py'
        data = source.encode()
        manifest[name] = hashlib.sha256(data).hexdigest()
        archive.writestr(name, data)
(out / 'source-manifest.json').write_text(json.dumps(manifest, indent=2) + '\n')
for filename in ['rip_campaign.py', 'rip_candidates.py', 'rip_wide_candidates.py',
                 'rip_rgba_candidates.py', 'rip_io_candidates.py']:
    data = subprocess.check_output(['git', 'show', f"{report['base_commit']}:benchmarks/{filename}"], cwd=root)
    (out / (filename + '.txt')).write_bytes(data)
assert hashlib.sha256((out / 'rip_campaign.py.txt').read_bytes()).hexdigest() == report['benchmark_sha256']
provenance = dict(base_commit=report['base_commit'], suite=report['suite'],
                  stopped_at_utc=report['stopped_at_utc'], memory_metric='tracemalloc, not complete RSS',
                  partial_source_folders_not_counted=partial_folders,
                  final_modules='Derived from verified last accepted snapshot, or initial if no acceptance',
                  zip_sha256=hashlib.sha256((out / 'source-snapshots.zip').read_bytes()).hexdigest())
(out / 'provenance.json').write_text(json.dumps(provenance, indent=2) + '\n')
prior = root / 'benchmarks/campaign_runs/2026-11-11-state-query/before-wide-timing'
prior.mkdir(parents=True, exist_ok=True)
shutil.copy2(review / 'rip-run/report.json', prior / 'report.json')
shutil.copy2(review / 'rip-progress.log', prior / 'progress.log')
prior_report = json.loads((prior / 'report.json').read_text())
prior_manifest = {}
with zipfile.ZipFile(prior / 'source-snapshots.zip', 'w', compression=zipfile.ZIP_DEFLATED) as archive:
    for folder in sorted((review / 'rip-run').glob('round-*')) + [review / 'rip-run/initial']:
        for k in FILES:
            path = folder / (k + '.py')
            data = path.read_bytes()
            name = str(path.relative_to(review / 'rip-run'))
            archive.writestr(name, data)
            prior_manifest[name] = hashlib.sha256(data).hexdigest()
(prior / 'source-manifest.json').write_text(json.dumps(prior_manifest, indent=2) + '\n')
shutil.copy2(__file__, out / 'audit.py.txt')
for directory in (out, prior):
    manifest_data = json.loads((directory / 'source-manifest.json').read_text())
    with zipfile.ZipFile(directory / 'source-snapshots.zip') as archive:
        assert set(archive.namelist()) == set(manifest_data)
        for name, expected in manifest_data.items():
            assert hashlib.sha256(archive.read(name)).hexdigest() == expected
(out / 'README.md').write_text(f'''# Fase final de RIP

Motivo real de parada: `{report['stop_reason']}`. Rodadas concluídas: {len(report['rounds'])}; aceites: {result['accepted']}; sequência final sem ganho: {count}; alvo: 25. Convergência alcançada: {result['convergence_reached']}. Rodadas parciais e erros de infraestrutura não contam. Não se soma a sequência anterior da suíte v2 nem a da versão 0.8.22.

Suíte v3: 30 PRTs de correção e 12 cenários cronometrados, incluindo a folha opaca de 6000 pontos. A ampliação evita testes inertes de políticas ativadas apenas acima de 4096 pontos. As três rodadas v2 anteriores ficam separadas. Os pares de hashes históricos impedem repetir variantes já testadas no mesmo líder.

A auditoria independente verificou histórico, reinícios do contador, hashes, checkout e 30 PRTs idênticos. O snapshot final foi derivado da última candidata aceita ou da base; isso não altera a flag de conclusão original do relatório. Os TXT preservam o driver e propostas medidos. O pico é tracemalloc, sem todo o RSS nativo. Os resultados não provam ótimo global para toda imagem ou máquina.
''')
print(json.dumps({k: v for k, v in result.items() if k != 'prt_sha256'}, indent=2))
